def quality_check(generated_image, original_image, user_request):
    """
    Автоматична перевірка якості згенерованого дизайну
    """
    checks = {
        'geometry_preserved': check_geometry(generated_image, original_image),
        'photorealistic': check_photorealism(generated_image),
        'proper_scale': check_scale(generated_image),
        'style_match': check_style(generated_image, user_request['style']),
        'functional': check_functionality(generated_image, user_request['room_type']),
        'budget_appropriate': check_budget(generated_image, user_request['budget_tier']),
        'high_quality': check_technical_quality(generated_image)
    }
    
    return checks
